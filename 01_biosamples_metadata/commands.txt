### Get all BioSamples as UIDs
esearch -db biosample -query "$taxids" |
    efetch -format uid > biosample_uids.txt

### Count how many UIDs there are
esearch -db biosample \
  -query "$taxids" |
grep '<Count>'

### Retrieve the BioSamples
esearch -db biosample \
  -query "$taxids" |
efetch -db biosample -format xml > biosamples.xml

### Count the retrieved BioSamples
grep -o 'accession="SAM[A-Z]*[0-9]*"' biosamples.xml | wc -l
grep -o 'accession="SAMN[0-9]*"' biosamples.xml | wc -l
grep -o 'accession="SAMEA[0-9]*"' biosamples.xml | wc -l
grep -o 'accession="SAMD[0-9]*"' biosamples.xml | wc -l
